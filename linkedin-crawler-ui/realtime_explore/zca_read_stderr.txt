    async def _read_stderr(self, state: ListenerState, proc: Any) -> None:
        if not proc.stderr:
            return
        while True:
            line = await proc.stderr.readline()
            if not line:
                return
            logger.warning(
                "ZCA listener stderr user={}: {}",
                state.user_id,
                line.decode("utf-8", errors="replace").strip()[:1000],
            )
